FastAPI Admin Panel Project

Requires completing API Patterns first. Build a product catalog admin panel with authentication, CRUD operations, file uploads, and real-time updates using FastAPI and vanilla JavaScript.

1

Project Overview

This project is a product catalog admin panel. Users log in with email and password. The dashboard shows a searchable product table with create, update, delete, and upload image functionality. Polling checks for new products every 30 seconds.

Files needed: Create app.py (FastAPI backend), index.html (admin panel), and products.db (SQLite database, auto-created).

FastAPI Backend: Models and Routes

The backend has User and Product models, JWT login, and full CRUD endpoints. It uses SQLite for persistence. All endpoints under /api/ return JSON.

python 
from fastapi import FastAPI, Depends, HTTPException, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session
from passlib.context import CryptContext
from datetime import datetime, timedelta
import jwt, os
from typing import Optional

#### Setup ####
app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

DATABASE_URL = "sqlite:///./products.db"
engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(bind=engine)
Base = declarative_base()

SECRET_KEY = "your-secret-key-change-this"
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

#### Models ####
class UserDB(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True)
    email = Column(String, unique=True, index=True)
    hashed_password = Column(String)

class ProductDB(Base):
    __tablename__ = "products"
    id = Column(Integer, primary_key=True)
    name = Column(String, index=True)
    price = Column(Float)
    inventory = Column(Integer)
    image_url = Column(String, nullable=True)

Base.metadata.create_all(bind=engine)

#### Schemas ####
class LoginRequest(BaseModel):
    email: str
    password: str

class LoginResponse(BaseModel):
    access_token: str

class Product(BaseModel):
    id: int
    name: str
    price: float
    inventory: int
    image_url: Optional[str] = None

Login Module

The login endpoint validates email and password, returns a JWT token. The token is sent in the Authorization header for protected endpoints.

python 
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

def hash_password(password: str):
    return pwd_context.hash(password)

def verify_password(plain, hashed):
    return pwd_context.verify(plain, hashed)

def create_token(user_id: int) -> str:
    payload = {
        "user_id": user_id,
        "exp": datetime.utcnow() + timedelta(hours=24)
    }
    return jwt.encode(payload, SECRET_KEY, algorithm="HS256")

def verify_token(token: str):
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=["HS256"])
        return payload["user_id"]
    except:
        raise HTTPException(status_code=401, detail="Invalid token")
python 
@app.post("/api/login")
def login(req: LoginRequest, db: Session = Depends(get_db)):
    user = db.query(UserDB).filter(UserDB.email == req.email).first()
    if not user or not verify_password(req.password, user.hashed_password):
        raise HTTPException(status_code=401, detail="Wrong credentials")
    token = create_token(user.id)
    return {"access_token": token}

Products Table and API Integration

Fetch products from the API and populate the table. Send the JWT token in the Authorization header. Handle loading state and errors.

python 
@app.get("/api/products")
def list_products(
    search: Optional[str] = None,
    skip: int = 0,
    limit: int = 100,
    authorization: str = Header(...),
    db: Session = Depends(get_db)
):
    user_id = verify_token(authorization.replace("Bearer ", ""))
    query = db.query(ProductDB)
    if search:
        query = query.filter(ProductDB.name.ilike(f"%{search}%"))
    return query.offset(skip).limit(limit).all()
javascript 
const loadProducts = async (search = "") => {
  const token = localStorage.getItem("token");
  const params = new URLSearchParams({ search });

  try {
    const response = await fetch("/api/products?" + params, {
      headers: { "Authorization": "Bearer " + token }
    });

    if (response.status === 401) {
      localStorage.removeItem("token");
      location.reload();
      return;
    }

    const products = await response.json();
    const tbody = document.querySelector("#productTable tbody");
    tbody.innerHTML = "";

    products.forEach(p => {
      const row = document.createElement("tr");
      row.innerHTML = `
        <td>${p.id}</td>
        <td>${p.name}</td>
        <td>$${p.price}</td>
        <td>${p.inventory}</td>
        <td class="actions">
          <button onclick="editProduct(${p.id})">Edit</button>
          <button onclick="deleteProduct(${p.id})">Delete</button>
        </td>
      `;
      tbody.appendChild(row);
    });
  } catch (err) {
    alert("Error loading products");
  }
};

Create and Update Products

POST creates new products. PATCH updates existing ones. Use a modal form to collect name, price, and inventory. Validate before sending.

python 
class ProductCreate(BaseModel):
    name: str
    price: float
    inventory: int

@app.post("/api/products")
def create_product(
    data: ProductCreate,
    authorization: str = Header(...),
    db: Session = Depends(get_db)
):
    verify_token(authorization.replace("Bearer ", ""))
    product = ProductDB(name=data.name, price=data.price, inventory=data.inventory)
    db.add(product)
    db.commit()
    db.refresh(product)
    return product

@app.patch("/api/products/{id}")
def update_product(
    id: int,
    data: ProductCreate,
    authorization: str = Header(...),
    db: Session = Depends(get_db)
):
    verify_token(authorization.replace("Bearer ", ""))
    product = db.query(ProductDB).filter(ProductDB.id == id).first()
    if not product:
        raise HTTPException(status_code=404)
    product.name = data.name
    product.price = data.price
    product.inventory = data.inventory
    db.commit()
    return product
javascript 
const createProduct = async (e) => {
  e.preventDefault();
  const token = localStorage.getItem("token");
  const name = document.getElementById("productName").value;
  const price = parseFloat(document.getElementById("productPrice").value);
  const inventory = parseInt(document.getElementById("productInventory").value);

  if (!name || !price || !inventory) {
    alert("All fields required");
    return;
  }

  const response = await fetch("/api/products", {
    method: "POST",
    headers: {
      "Authorization": "Bearer " + token,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({ name, price, inventory })
  });

  if (response.ok) {
    loadProducts();
    e.target.reset();
  } else {
    alert("Create failed");
  }
};

Delete Products

DELETE removes a product. Confirm with the user before deleting. The backend checks authorization before allowing deletion.

python 
@app.delete("/api/products/{id}")
def delete_product(
    id: int,
    authorization: str = Header(...),
    db: Session = Depends(get_db)
):
    verify_token(authorization.replace("Bearer ", ""))
    product = db.query(ProductDB).filter(ProductDB.id == id).first()
    if not product:
        raise HTTPException(status_code=404)
    db.delete(product)
    db.commit()
    return {"success": True}
javascript 
const deleteProduct = async (id) => {
  if (!confirm("Delete this product?")) return;

  const token = localStorage.getItem("token");
  const response = await fetch("/api/products/" + id, {
    method: "DELETE",
    headers: { "Authorization": "Bearer " + token }
  });

  if (response.ok) {
    loadProducts();
  } else {
    alert("Delete failed");
  }
};

Search and Filter

The search input filters products by name. Use a debounce to avoid too many requests while typing. Send the query parameter to the API.

javascript 
let searchTimeout;

document.getElementById("searchInput").addEventListener("input", (e) => {
  clearTimeout(searchTimeout);
  searchTimeout = setTimeout(() => {
    loadProducts(e.target.value);  // wait 300ms before searching
  }, 300);
});

document.getElementById("refreshBtn").addEventListener("click", () => {
  document.getElementById("searchInput").value = "";
  loadProducts();
});

File Uploads for Product Images

Use FormData to upload image files alongside product data. The backend saves the file and stores the filename in the database.

python 
@app.post("/api/products/{id}/upload")
async def upload_image(
    id: int,
    file: UploadFile = File(...),
    authorization: str = Header(...),
    db: Session = Depends(get_db)
):
    verify_token(authorization.replace("Bearer ", ""))
    product = db.query(ProductDB).filter(ProductDB.id == id).first()
    if not product:
        raise HTTPException(status_code=404)

    filename = f"product_{id}_{file.filename}"
    with open(f"uploads/{filename}", "wb") as f:
        content = await file.read()
        f.write(content)

    product.image_url = filename
    db.commit()
    return product
javascript 
const uploadImage = async (productId, fileInput) => {
  const file = fileInput.files[0];
  if (!file) return;

  const token = localStorage.getItem("token");
  const formData = new FormData();
  formData.append("file", file);

  const response = await fetch("/api/products/" + productId + "/upload", {
    method: "POST",
    headers: { "Authorization": "Bearer " + token },
    body: formData
  });

  if (response.ok) {
    alert("Image uploaded");
    loadProducts();
  }
};

Polling for New Products

Set an interval to check for new products every 30 seconds. This keeps the dashboard synchronized if another admin adds products. Stop polling on logout.

javascript 
let pollInterval;

const startPolling = () => {
  pollInterval = setInterval(() => {
    loadProducts();
  }, 30000);  // poll every 30 seconds
};

const stopPolling = () => {
  clearInterval(pollInterval);
};

document.getElementById("logoutBtn").addEventListener("click", () => {
  localStorage.removeItem("token");
  stopPolling();
  location.reload();
});

// Start polling when dashboard loads
if (localStorage.getItem("token")) {
  startPolling();
}
