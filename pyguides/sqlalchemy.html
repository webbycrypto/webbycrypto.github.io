Project Setup & Connection Strings

SQLAlchemy supports many databases. The connection string tells it which one to use and where to find it. SQLite is a single file - perfect for local development. PostgreSQL is used in production. You do not need to install anything extra for SQLite; it is built into Python.

python 
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, DeclarativeBase

# SQLite - stores everything in a single file, no server needed
DATABASE_URL = "sqlite:///./app.db"

# PostgreSQL - swap this in for production
# DATABASE_URL = "postgresql://user:password@localhost:5432/mydb"

# MySQL
# DATABASE_URL = "mysql+pymysql://user:password@localhost:3306/mydb"

# echo=True prints every SQL statement to the terminal - useful for debugging
engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False}, echo=False)

# SessionLocal is a factory - call it to get a new database session
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# All models inherit from Base - SQLAlchemy uses this to track your tables
class Base(DeclarativeBase):
    pass

Defining Models

A model is a Python class that maps to a database table. Each class attribute decorated with Column becomes a column in the table. SQLAlchemy handles the SQL - you just work with Python objects.

python 
from sqlalchemy import Column, Integer, String, Boolean, DateTime, Float, Text
from sqlalchemy.sql import func
from database import Base

class User(Base):
    __tablename__ = "users"  # name of the table in the database

    id         = Column(Integer, primary_key=True, index=True)  # auto-incrementing ID
    email      = Column(String(120), unique=True, nullable=False, index=True)
    name       = Column(String(100))
    is_active  = Column(Boolean, default=True)
    created_at = Column(DateTime(timezone=True), server_default=func.now())  # set by the DB

    def __repr__(self):
        # What Python prints when you do print(user) or look at a list of users
        return f"<User id={self.id} email={self.email}>"

class Product(Base):
    __tablename__ = "products"

    id          = Column(Integer, primary_key=True, index=True)
    name        = Column(String(200), nullable=False)
    description = Column(Text)
    price       = Column(Float, nullable=False)
    in_stock    = Column(Boolean, default=True)

    def __repr__(self):
        return f'<Product id={self.id} name={self.name} price={self.price}>'

Creating Tables & Sessions

create_all() looks at all models that inherit from Base and creates their tables if they don't exist yet. The get_db() function opens a session, hands it to your code, then closes it automatically - even if something goes wrong.

python 
from database import engine, SessionLocal, Base
import models  # importing models registers them with Base

# Create all tables that don't exist yet - safe to call every startup
Base.metadata.create_all(bind=engine)

# --- Session as a context manager ---
# This is the cleanest pattern: the session is always closed, even on errors

def get_db():
    db = SessionLocal()       # open a new session
    try:
        yield db              # hand it to the caller
    finally:
        db.close()            # always close, no matter what happened

# Using it directly (without FastAPI/Flask dependency injection):
if __name__ == "__main__":
    db = SessionLocal()
    try:
        print("Tables created. Session open.")
        # ... do database work here ...
    finally:
        db.close()

Create & Read

Adding a record takes three steps: create the object, add it to the session, commit. Reading uses session.query() to build a SELECT statement - filter() adds a WHERE clause, first() returns one result, all() returns a list.

python 
from database import SessionLocal
from models import User

db = SessionLocal()

# --- CREATE ---
new_user = User(email="alice@example.com", name="Alice")
db.add(new_user)        # stage the insert
db.commit()             # write to the database
db.refresh(new_user)    # reload the object so new_user.id is populated
print(new_user)         # <User id=1 email=alice@example.com>

# Add multiple records at once
db.add_all([
    User(email="bob@example.com",     name="Bob"),
    User(email="charlie@example.com", name="Charlie"),
])
db.commit()

# --- READ ---
# Get all users
all_users = db.query(User).all()
print(all_users)  # [<User ...>, <User ...>, ...]

# Get by primary key
user = db.query(User).filter(User.id == 1).first()
print(user.name)  # Alice

# Get by email (or any column)
user = db.query(User).filter(User.email == "bob@example.com").first()

# Filter with multiple conditions (AND)
active_bobs = db.query(User).filter(User.name == "Bob", User.is_active == True).all()

# Count results
count = db.query(User).count()
print(f"{count} users in the database")

# Order and limit
top5 = db.query(User).order_by(User.created_at.desc()).limit(5).all()

db.close()

Update & Delete

To update a record, fetch it, change the attribute, then commit. To delete, fetch it and call session.delete(). The bulk update pattern lets you update many rows at once without loading them all into Python first.

python 
from database import SessionLocal
from models import User

db = SessionLocal()

# --- UPDATE (fetch, change, commit) ---
user = db.query(User).filter(User.id == 1).first()
if user:
    user.name = "Alice Updated"  # change the attribute directly
    db.commit()                  # write the change
    db.refresh(user)             # reload to confirm
    print(user.name)             # Alice Updated

# --- BULK UPDATE (without loading each object) ---
db.query(User).filter(User.is_active == False).update({"name": "Inactive"})
db.commit()

# --- DELETE (fetch, delete, commit) ---
user = db.query(User).filter(User.id == 2).first()
if user:
    db.delete(user)  # mark for deletion
    db.commit()      # execute the DELETE

# --- BULK DELETE ---
db.query(User).filter(User.email.like("%spam%")).delete(synchronize_session=False)
db.commit()

# --- SAFE DELETE PATTERN ---
# Check it exists before deleting - avoids silent failures
def delete_user(db, user_id: int) -> bool:
    user = db.query(User).filter(User.id == user_id).first()
    if not user:
        return False        # nothing to delete
    db.delete(user)
    db.commit()
    return True             # deleted successfully

db.close()

Relationships

A relationship links two tables together. ForeignKey creates the database-level link; relationship() creates the Python-level link so you can navigate from a parent object to its children (and back) without writing any SQL.

python 
from sqlalchemy import Column, Integer, String, ForeignKey, Text
from sqlalchemy.orm import relationship
from database import Base

class Author(Base):
    __tablename__ = "authors"

    id    = Column(Integer, primary_key=True)
    name  = Column(String(100), nullable=False)

    # 'books' is a Python attribute - SQLAlchemy loads the related Book rows automatically
    books = relationship("Book", back_populates="author", cascade="all, delete-orphan")
    # cascade="all, delete-orphan" means: if you delete an Author, delete their Books too

    def __repr__(self):
        return f"<Author {self.name}>"

class Book(Base):
    __tablename__ = "books"

    id        = Column(Integer, primary_key=True)
    title     = Column(String(200), nullable=False)
    body      = Column(Text)
    author_id = Column(Integer, ForeignKey("authors.id"), nullable=False)  # the FK column

    # 'author' is a Python attribute - gives you back the Author object
    author = relationship("Author", back_populates="books")

    def __repr__(self):
        return f"<Book '{self.title}'>"
python 
from database import SessionLocal, engine, Base
import models_related as m

Base.metadata.create_all(bind=engine)
db = SessionLocal()

# Create an author with books in one go
author = m.Author(name="George Orwell", books=[
    m.Book(title="1984"),
    m.Book(title="Animal Farm"),
])
db.add(author)
db.commit()
db.refresh(author)

# Navigate from author → books
for book in author.books:
    print(book.title)   # 1984 / Animal Farm

# Navigate from book → author
book = db.query(m.Book).filter(m.Book.title == "1984").first()
print(book.author.name)  # George Orwell

db.close()

Alembic Migrations

Alembic tracks changes to your models and generates SQL migration scripts. When you add a column or change a table, you create a new migration rather than deleting and recreating the database - this keeps your existing data safe.

bash 
# 1. Initialise Alembic - creates an alembic/ folder and alembic.ini
alembic init alembic

# 2. Edit alembic/env.py - tell it about your models and database URL
#    (see env.py below)

# 3. Auto-generate a migration by comparing your models to the current DB
alembic revision --autogenerate -m "add users table"

# 4. Apply the migration to the database
alembic upgrade head

# Other useful commands:
alembic history          # list all migrations
alembic current          # show which migration the DB is at
alembic downgrade -1     # roll back one migration
python 
# Add these two lines near the top of env.py (after the default imports):
from database import Base, DATABASE_URL   # import your Base and URL
import models                              # import your models so Alembic sees them

# Then update the two config lines Alembic looks for:
config.set_main_option("sqlalchemy.url", DATABASE_URL)  # tell Alembic the DB URL
target_metadata = Base.metadata                          # tell Alembic about your tables

# Everything else in env.py stays the same - Alembic auto-generates the rest.

Async SQLAlchemy

The async version of SQLAlchemy works with asyncio - useful inside FastAPI async routes or any async Python code. The API is nearly identical to the sync version; the main differences are using `await` and `AsyncSession`.

python 
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy.orm import DeclarativeBase

# Note: aiosqlite driver is required for async SQLite
ASYNC_DATABASE_URL = "sqlite+aiosqlite:///./app_async.db"

# For async PostgreSQL use asyncpg:
# ASYNC_DATABASE_URL = "postgresql+asyncpg://user:password@localhost/mydb"

async_engine = create_async_engine(ASYNC_DATABASE_URL, echo=False)

AsyncSessionLocal = async_sessionmaker(
    bind=async_engine,
    class_=AsyncSession,
    expire_on_commit=False,  # keeps objects usable after commit without a reload
)

class Base(DeclarativeBase):
    pass

# Async version of get_db() - use with 'async with' or as a FastAPI dependency
async def get_async_db():
    async with AsyncSessionLocal() as session:
        yield session
python 
import asyncio
from sqlalchemy import select
from database_async import async_engine, AsyncSessionLocal, Base
from models import User  # same models work for both sync and async

async def main():
    # Create tables
    async with async_engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    async with AsyncSessionLocal() as db:
        # CREATE
        user = User(email="async@example.com", name="Async User")
        db.add(user)
        await db.commit()
        await db.refresh(user)
        print(user)

        # READ - use select() instead of db.query() for async
        result = await db.execute(select(User).where(User.email == "async@example.com"))
        found = result.scalar_one_or_none()  # returns the object or None
        print(found)

        # UPDATE
        found.name = "Updated Async User"
        await db.commit()

        # DELETE
        await db.delete(found)
        await db.commit()

asyncio.run(main())

SQLAlchemy with FastAPI / Flask / Django / XRPL

Each framework wires SQLAlchemy in slightly differently. FastAPI uses dependency injection, Flask uses an extension, Django typically uses its own ORM but you can add SQLAlchemy alongside it, and XRPL apps use it as a local cache for wallet and transaction data.

python 
# FastAPI uses Depends() to inject a database session into each route.
# The session is opened at the start of the request and closed at the end.

from fastapi import FastAPI, Depends
from sqlalchemy.orm import Session
from database import engine, Base, get_db
import models

Base.metadata.create_all(bind=engine)
app = FastAPI()

@app.post("/users")
def create_user(email: str, db: Session = Depends(get_db)):
    # db is a fresh session for this request - injected automatically by FastAPI
    user = models.User(email=email)
    db.add(user)
    db.commit()
    db.refresh(user)
    return {"id": user.id, "email": user.email}

@app.get("/users/{user_id}")
def get_user(user_id: int, db: Session = Depends(get_db)):
    user = db.query(models.User).filter(models.User.id == user_id).first()
    if not user:
        raise HTTPException(status_code=404, detail="User not found")
    return {"id": user.id, "email": user.email}
python 
# Flask uses the flask-sqlalchemy extension which wraps SQLAlchemy.
# Models inherit from db.Model instead of Base, and db.session replaces SessionLocal.

from flask import Flask
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config["SQLALCHEMY_DATABASE_URI"] = "sqlite:///app.db"
app.config["SQLALCHEMY_TRACK_MODIFICATIONS"] = False

db = SQLAlchemy(app)

class User(db.Model):
    id    = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(120), unique=True, nullable=False)
    name  = db.Column(db.String(100))

with app.app_context():
    db.create_all()  # create tables

@app.route("/users", methods=["POST"])
def create_user():
    from flask import request, jsonify
    data = request.get_json()
    user = User(email=data["email"], name=data.get("name"))
    db.session.add(user)
    db.session.commit()
    return jsonify({"id": user.id, "email": user.email}), 201

@app.route("/users/<int:user_id>")
def get_user(user_id):
    from flask import jsonify
    user = User.query.get_or_404(user_id)
    return jsonify({"id": user.id, "email": user.email})

# Django has its own ORM and uses it by default.
# You can add a raw SQLAlchemy session alongside Django for edge cases
# (e.g. complex queries, legacy databases, or shared code with non-Django apps).

# settings.py - Django still manages its own DB connection
DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.sqlite3",
        "NAME": BASE_DIR / "db.sqlite3",
    }
}

# db_alchemy.py - separate SQLAlchemy session for raw queries
from sqlalchemy import create_engine, text
from sqlalchemy.orm import sessionmaker

# Point it at the same SQLite file Django uses
engine  = create_engine("sqlite:///./db.sqlite3")
Session = sessionmaker(bind=engine)

def raw_query_example():
    # Use this only when Django ORM can't express what you need
    with Session() as session:
        result = session.execute(text("SELECT * FROM myapp_mymodel WHERE active = 1"))
        return result.fetchall()

# views.py - using Django ORM normally (recommended for most cases)
from django.shortcuts import get_object_or_404
from .models import MyModel

def my_view(request, pk):
    obj = get_object_or_404(MyModel, pk=pk)
    return obj
python 
# XRPL apps use SQLAlchemy to cache wallet data and transaction history locally.
# This avoids re-querying the ledger every time and lets you search/filter offline.

from sqlalchemy import create_engine, Column, Integer, String, DateTime
from sqlalchemy.orm import sessionmaker, DeclarativeBase
from datetime import datetime

engine = create_engine("sqlite:///xrpl_cache.db")
Session = sessionmaker(bind=engine)

class Base(DeclarativeBase):
    pass

class Wallet(Base):
    __tablename__ = "wallets"
    id         = Column(Integer, primary_key=True)
    address    = Column(String(35), unique=True, nullable=False)  # classic address rXXX...
    label      = Column(String(100))  # a friendly name you give the wallet
    created_at = Column(DateTime, default=datetime.utcnow)

class Transaction(Base):
    __tablename__ = "transactions"
    id          = Column(Integer, primary_key=True)
    tx_hash     = Column(String(64), unique=True, nullable=False)
    tx_type     = Column(String(50))   # Payment, EscrowCreate, NFTokenMint, etc.
    from_addr   = Column(String(35))
    to_addr     = Column(String(35))
    amount_drops= Column(String(20))   # store as string to avoid float precision issues
    ledger_date = Column(DateTime)
    raw_json    = Column(String)       # store the full tx JSON for reference

Base.metadata.create_all(bind=engine)

# --- Example: cache a transaction after submitting it ---
def cache_transaction(tx_hash, tx_type, from_addr, to_addr, amount_drops, raw_json):
    with Session() as db:
        tx = Transaction(
            tx_hash=tx_hash,
            tx_type=tx_type,
            from_addr=from_addr,
            to_addr=to_addr,
            amount_drops=str(amount_drops),
            raw_json=raw_json,
            ledger_date=datetime.utcnow(),
        )
        db.add(tx)
        db.commit()

# --- Example: look up transaction history from local cache ---
def get_wallet_history(address: str):
    with Session() as db:
        return db.query(Transaction).filter(
            (Transaction.from_addr == address) | (Transaction.to_addr == address)
        ).order_by(Transaction.ledger_date.desc()).all()
